# Django 중고거래 프로젝트 - 상품 게시판, 찜하기, 실시간 채팅

## 이번 수업의 목표

이번 시간에는 앞에서 만든 회원가입 / 로그인 기능을 기반으로 중고거래 서비스의 핵심 기능을 구현합니다.

완성 기능:

1. 로그인한 사용자만 판매글 작성
2. 제목, 본문, 사진, 판매 지역, 판매 가격, 네고 가능 여부 저장
3. 조회수 표시
4. 찜하기 / 찜 취소
5. 상품별 채팅 시작 인원 표시
6. 본인 글만 수정 / 삭제
7. 판매자와 1:1 채팅방 생성
8. Django Channels + WebSocket 실시간 채팅
9. 채팅 메시지 Supabase PostgreSQL 저장
10. 상품 이미지 Supabase Storage 저장

# 1. Product / Wishlist / Chat 모델 만들기

`market/models.py`의 기존 `Profile` 아래에 추가합니다.

```python
from django.db import models
from django.contrib.auth.models import User


class Product(models.Model):

    seller = models.ForeignKey(
        User,
        on_delete=models.CASCADE,
        related_name="products"
    )

    title = models.CharField(
        max_length=100
    )

    content = models.TextField()

    image_url = models.URLField(
        blank=True,
        null=True
    )

    region = models.CharField(
        max_length=100
    )

    price = models.PositiveIntegerField()

    negotiable = models.BooleanField(
        default=False
    )

    view_count = models.PositiveIntegerField(
        default=0
    )

    created_at = models.DateTimeField(
        auto_now_add=True
    )

    updated_at = models.DateTimeField(
        auto_now=True
    )

    class Meta:
        ordering = ["-created_at"]

    def __str__(self):
        return self.title

    @property
    def wish_count(self):
        return self.wishlists.count()

    @property
    def chat_count(self):
        return self.chat_rooms.count()


class Wishlist(models.Model):

    user = models.ForeignKey(
        User,
        on_delete=models.CASCADE,
        related_name="wishlists"
    )

    product = models.ForeignKey(
        Product,
        on_delete=models.CASCADE,
        related_name="wishlists"
    )

    created_at = models.DateTimeField(
        auto_now_add=True
    )

    class Meta:
        constraints = [
            models.UniqueConstraint(
                fields=["user", "product"],
                name="unique_user_product_wishlist"
            )
        ]

    def __str__(self):
        return f"{self.user.username} - {self.product.title}"


class ChatRoom(models.Model):

    product = models.ForeignKey(
        Product,
        on_delete=models.CASCADE,
        related_name="chat_rooms"
    )

    seller = models.ForeignKey(
        User,
        on_delete=models.CASCADE,
        related_name="selling_chat_rooms"
    )

    buyer = models.ForeignKey(
        User,
        on_delete=models.CASCADE,
        related_name="buying_chat_rooms"
    )

    created_at = models.DateTimeField(
        auto_now_add=True
    )

    class Meta:
        constraints = [
            models.UniqueConstraint(
                fields=["product", "buyer"],
                name="unique_product_buyer_chatroom"
            )
        ]

    def __str__(self):
        return f"{self.product.title} / {self.seller.username} - {self.buyer.username}"


class ChatMessage(models.Model):

    room = models.ForeignKey(
        ChatRoom,
        on_delete=models.CASCADE,
        related_name="messages"
    )

    sender = models.ForeignKey(
        User,
        on_delete=models.CASCADE,
        related_name="chat_messages"
    )

    content = models.TextField()

    created_at = models.DateTimeField(
        auto_now_add=True
    )

    class Meta:
        ordering = ["created_at"]

    def __str__(self):
        return f"{self.sender.username}: {self.content[:20]}"
```

관계는 다음처럼 이해하면 됩니다.

```text
User 1 : N Product

User 1 : N Wishlist
Product 1 : N Wishlist

Product 1 : N ChatRoom
ChatRoom 1 : N ChatMessage
```

찜 횟수와 채팅 시작 인원은 숫자를 따로 저장하지 않고 실제 관계 데이터 개수를 계산합니다.

```python
product.wishlists.count()
product.chat_rooms.count()
```

# 5. Supabase Storage 업로드 함수

`market/storage.py` 생성:

```python
import os
import uuid

from supabase import create_client


SUPABASE_URL = os.getenv("SUPABASE_URL")
SUPABASE_KEY = os.getenv("SUPABASE_SERVICE_ROLE_KEY")

BUCKET_NAME = "product-images"

supabase = create_client(
    SUPABASE_URL,
    SUPABASE_KEY
)


def upload_product_image(uploaded_file, user_id):

    extension = (
        uploaded_file.name
        .split(".")[-1]
        .lower()
    )

    file_name = (
        f"{uuid.uuid4().hex}.{extension}"
    )

    file_path = (
        f"products/{user_id}/{file_name}"
    )

    supabase.storage.from_(
        BUCKET_NAME
    ).upload(
        path=file_path,
        file=uploaded_file.read(),
        file_options={
            "content-type": uploaded_file.content_type,
            "upsert": "false",
        }
    )

    return (
        supabase.storage
        .from_(BUCKET_NAME)
        .get_public_url(file_path)
    )
```

동작:

```text
사용자 이미지 선택
       ↓
Django
       ↓
Supabase Storage
       ↓
Public URL 생성
       ↓
Product.image_url 저장
```

# 9. 상품 등록

```python
@login_required
def product_create(request):

    if request.method == "POST":

        form = ProductForm(
            request.POST,
            request.FILES
        )

        if form.is_valid():

            product = form.save(
                commit=False
            )

            product.seller = request.user

            image = form.cleaned_data.get(
                "image"
            )

            if image:

                product.image_url = (
                    upload_product_image(
                        image,
                        request.user.id
                    )
                )

            product.save()

            return redirect(
                "product_detail",
                product_id=product.id
            )

    else:

        form = ProductForm()

    return render(
        request,
        "market/product_form.html",
        {
            "form": form,
            "page_title": "중고상품 등록",
        }
    )
```

# 10. 상품 상세 + 조회수

```python
def product_detail(request, product_id):

    product = get_object_or_404(
        Product,
        id=product_id
    )

    Product.objects.filter(
        id=product.id
    ).update(
        view_count=F("view_count") + 1
    )

    product.refresh_from_db(
        fields=["view_count"]
    )

    is_wished = False

    if request.user.is_authenticated:

        is_wished = (
            Wishlist.objects.filter(
                user=request.user,
                product=product
            ).exists()
        )

    return render(
        request,
        "market/product_detail.html",
        {
            "product": product,
            "is_wished": is_wished,
        }
    )
```

현재 수업에서는 상세 페이지를 열 때마다 조회수가 1씩 증가하는 단순한 방식입니다.

# 13. 찜하기 / 찜 취소

```python
@login_required
def wishlist_toggle(request, product_id):

    product = get_object_or_404(
        Product,
        id=product_id
    )

    if product.seller == request.user:

        return redirect(
            "product_detail",
            product_id=product.id
        )

    if request.method == "POST":

        wishlist, created = (
            Wishlist.objects
            .get_or_create(
                user=request.user,
                product=product
            )
        )

        if not created:
            wishlist.delete()

    return redirect(
        "product_detail",
        product_id=product.id
    )
```

동작:

```text
아직 찜 안함
→ Wishlist 생성

이미 찜함
→ Wishlist 삭제
```

```python
def chat_start(request, product_id):

        Product,
        id=product_id

    if product.seller == request.user:

        return redirect(
            "product_detail",
```

# 16. 채팅방 화면 View

```python
@login_required
def chat_room(request, room_id):

    room = get_object_or_404(
        ChatRoom.objects.select_related(
            "product",
            "seller",
            "buyer"
        ),
        id=room_id
    )

    if (
        request.user != room.seller
        and
        request.user != room.buyer
    ):
        raise PermissionDenied

    chat_messages = (
        room.messages
        .select_related("sender")
        .all()
    )

    return render(
        request,
        "market/chat_room.html",
        {
            "room": room,
            "chat_messages": chat_messages,
        }
    )
```

# 25. ChatConsumer

`market/consumers.py`

```python
from asgiref.sync import async_to_sync

from channels.generic.websocket import (
    JsonWebsocketConsumer,
)

from django.db.models import Q
from django.utils import timezone

from .models import (
    ChatRoom,
    ChatMessage,
)


class ChatConsumer(
    JsonWebsocketConsumer
):

    def connect(self):

        self.room_id = (
            self.scope[
                "url_route"
            ]["kwargs"]["room_id"]
        )

        self.user = self.scope["user"]

        if self.user.is_anonymous:

            self.close()
            return

        room_exists = (
            ChatRoom.objects
            .filter(id=self.room_id)
            .filter(
                Q(seller=self.user)
                |
                Q(buyer=self.user)
            )
            .exists()
        )

        if not room_exists:

            self.close()
            return

        self.room_group_name = (
            f"chat-{self.room_id}"
        )

        async_to_sync(
            self.channel_layer.group_add
        )(
            self.room_group_name,
            self.channel_name
        )

        self.accept()


    def disconnect(
        self,
        close_code
    ):

        if hasattr(
            self,
            "room_group_name"
        ):

            async_to_sync(
                self.channel_layer.group_discard
            )(
                self.room_group_name,
                self.channel_name
            )


    def receive_json(
        self,
        content,
        **kwargs
    ):

        message_text = (
            content
            .get("message", "")
            .strip()
        )

        if not message_text:
            return

        room = ChatRoom.objects.get(
            id=self.room_id
        )

        message = ChatMessage.objects.create(
            room=room,
            sender=self.user,
            content=message_text
        )

        async_to_sync(
            self.channel_layer.group_send
        )(
            self.room_group_name,
            {
                "type": "chat.message",
                "message": message.content,
                "username": self.user.username,
                "created_at":
                    timezone.localtime(
                        message.created_at
                    ).strftime("%H:%M"),
            }
        )


    def chat_message(
        self,
        event
    ):

        self.send_json({
            "message": event["message"],
            "username": event["username"],
            "created_at": event["created_at"],
        })
```

메시지를 받을 때:

```python
ChatMessage.objects.create(...)
```

가 실행되므로 실시간 전달뿐 아니라 Supabase PostgreSQL에도 기록됩니다.

# 28. 실시간 채팅 HTML

```html
<!DOCTYPE html>
<html lang="ko">

<head>
    <meta charset="UTF-8">
    <title>채팅</title>

<body>

<h1>

</h1>

<p>

</p>

<a href="{% url 'chat_list' %}">
    채팅 목록
</a>

<hr>

<div
    id="chat-log"
    style="
        border: 1px solid black;
        width: 600px;
        height: 400px;
        overflow-y: auto;
        padding: 10px;
    "
>

    {% for message in chat_messages %}

        <div>

            <strong>

            </strong>

            <small>

</div>

    type="text"
    placeholder="메시지를 입력하세요."
    style="width: 500px;"
>

<button id="chat-message-submit">
    전송
</button>


<script>

const roomId = "{{ room.id }}";

const protocol =
    window.location.protocol === "https:"
        ? "wss"
        : "ws";

const chatSocket = new WebSocket(
    `${protocol}://${window.location.host}/ws/chat/${roomId}/`
);

const chatLog =
    document.getElementById("chat-log");

const messageInput =
    document.getElementById(
        "chat-message-input"
    );

const sendButton =
    document.getElementById(
        "chat-message-submit"
    );


chatSocket.onmessage = function(event) {

    const data =
        JSON.parse(event.data);

    const line =
        document.createElement("div");

    const name =
        document.createElement("strong");

    name.textContent =
        data.username;

    const text =
        document.createTextNode(
            " : "
            + data.message
            + " "
        );

    const time =
        document.createElement("small");

    time.textContent =
        data.created_at;

    line.appendChild(name);
    line.appendChild(text);
    line.appendChild(time);

    chatLog.appendChild(line);

    chatLog.scrollTop =
        chatLog.scrollHeight;
};


function sendMessage() {

    const message =
        messageInput.value.trim();

    if (!message) {
        return;
    }

    chatSocket.send(
        JSON.stringify({
            "message": message
        })
    );

    messageInput.value = "";
}


sendButton.addEventListener(
    "click",
    sendMessage
);


messageInput.addEventListener(
    "keydown",
    function(event) {

        if (event.key === "Enter") {
            sendMessage();
        }

    }
);

</script>

</body>

</html>
```

# 29. 채팅 테스트

실시간 채팅은 계정 두 개로 테스트하는 것이 가장 쉽습니다.

## 사용자 A

일반 Chrome:

```text
seller01 로그인
→ 상품 작성
```

## 사용자 B

Chrome 시크릿 창:

```text
buyer01 로그인
→ seller01 상품 접속
→ 찜하기
→ 판매자와 채팅하기
```

그 다음 seller01 계정에서도:

```text
내 채팅
→ buyer01 채팅방
```

으로 들어갑니다.

양쪽에서 메시지가 실시간으로 보이면 성공입니다.

새로고침했을 때 기존 메시지가 다시 보인다면 DB 저장도 성공한 것입니다.

같은 구매자가 같은 상품에 여러 번 채팅하기를 눌러도 ChatRoom은 하나만 유지됩니다.

# 35. 로컬과 실제 배포의 차이
